RESERVED — Firestore security rule blocks removed from firestore.rules when the
Data Recorder module was pulled from the active app. Paste these back into the
`match /databases/{database}/documents { ... }` block (they previously sat
right after the `equipmentControl` block) if the module is restored.

    // Data Recorder — calibration raw-data sheets. Append-only audit records:
    // never updated; corrections are new "amendment" documents referencing the
    // original sheet's ID, cancellation is a void marker (below). Hard delete
    // is ADMIN-ONLY by owner decision (see docs/DATA_RECORDER_DESIGN.md §6b).
    match /rawDataSheets/{sheetId} {
      allow read, create: if request.auth != null;
      allow update: if false;
      allow delete: if request.auth != null &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // Analysis FormulaSets — editable relative-error/uncertainty-budget
    // formulas per sheetType (docs/FORMULA_ENGINE_DESIGN.md). Per design §2b,
    // versioning and admin-only write are DEFERRED during initial development:
    // this is one mutable document per sheetType, not append-only yet.
    match /analysisFormulaSets/{sheetType} {
      allow read, write: if request.auth != null;
    }

    // Sheet Type Definitions — admin-editable header fields, series layout,
    // and env-round count per sheetType, generalizing the recorder beyond
    // the fixed force/ISO 7500-1 shape. Same deferred-versioning posture as
    // analysisFormulaSets above: one mutable document per sheetType.
    match /sheetTypeDefinitions/{sheetType} {
      allow read, write: if request.auth != null;
    }

    // Void markers: append-only cancellation records referencing a sheet by ID.
    // "Deleting" a sheet = creating one of these; the sheet itself is untouched.
    // Admin delete allowed so hard-deleting a sheet can clean up its marker.
    match /rawDataSheetVoids/{voidId} {
      allow read, create: if request.auth != null;
      allow update: if false;
      allow delete: if request.auth != null &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }
